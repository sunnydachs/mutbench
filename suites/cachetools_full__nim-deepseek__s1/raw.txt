import pytest

class TestCache:
    def test_init_negative_maxsize(self):
        with pytest.raises(ValueError):
            cachetools.Cache(-1)

    def test_init_zero_maxsize(self):
        c = cachetools.Cache(0)
        assert c.maxsize == 0
        assert c.currsize == 0

    def test_setitem_zero_maxsize(self):
        c = cachetools.Cache(0)
        with pytest.raises(ValueError):
            c["a"] = 1

    def test_setitem_getitem(self):
        c = cachetools.Cache(10)
        c["a"] = 1
        assert c["a"] == 1
        assert c.currsize == 1

    def test_getitem_missing(self):
        c = cachetools.Cache(10)
        with pytest.raises(KeyError):
            c["missing"]

    def test_contains(self):
        c = cachetools.Cache(10)
        assert "a" not in c
        c["a"] = 1
        assert "a" in c

    def test_len(self):
        c = cachetools.Cache(10)
        assert len(c) == 0
        c["a"] = 1
        assert len(c) == 1

    def test_iter(self):
        c = cachetools.Cache(10)
        c["a"] = 1
        c["b"] = 2
        assert set(iter(c)) == {"a", "b"}

    def test_delitem(self):
        c = cachetools.Cache(10)
        c["a"] = 1
        del c["a"]
        assert "a" not in c
        assert c.currsize == 0

    def test_delitem_missing(self):
        c = cachetools.Cache(10)
        with pytest.raises(KeyError):
            del c["missing"]

    def test_get_default(self):
        c = cachetools.Cache(10)
        assert c.get("a") is None
        assert c.get("a", 42) == 42
        c["a"] = 1
        assert c.get("a") == 1

    def test_pop(self):
        c = cachetools.Cache(10)
        c["a"] = 1
        assert c.pop("a") == 1
        assert "a" not in c

    def test_pop_missing_no_default(self):
        c = cachetools.Cache(10)
        with pytest.raises(KeyError):
            c.pop("a")

    def test_pop_missing_default(self):
        c = cachetools.Cache(10)
        assert c.pop("a", 42) == 42

    def test_setdefault(self):
        c = cachetools.Cache(10)
        assert c.setdefault("a", 1) == 1
        assert c["a"] == 1
        assert c.setdefault("a", 2) == 1

    def test_clear(self):
        c = cachetools.Cache(10)
        c["a"] = 1
        c["b"] = 2
        c.clear()
        assert len(c) == 0
        assert c.currsize == 0

    def test_getsizeof_custom(self):
        c = cachetools.Cache(10, getsizeof=lambda v: v)
        c["a"] = 3
        assert c.currsize == 3

    def test_getsizeof_negative(self):

        with pytest.raises(ValueError):
            c["a"] = 1

    def test_getsizeof_too_large(self):
        c = cachetools.Cache(5, getsizeof=lambda v: v)
        with pytest.raises(ValueError):
            c["a"] = 10

    def test_getsizeof_eviction(self):
        c = cachetools.Cache(5, getsizeof=lambda v: v)
        c["a"] = 3
        c["b"] = 3
        assert c.currsize <= 5

    def test_getsizeof_update_existing(self):
        c = cachetools.Cache(10, getsizeof=lambda v: v)
        c["a"] = 3
        c["a"] = 5
        assert c.currsize == 5

    def test_repr(self):

        assert "Cache" in repr(c)

        time[0] = 20.0
        assert "a" not in c

class TestCached:
    def test_basic(self):
        cache = cachetools.Cache(10)
        calls = []

        @cachetools.cached(cache)
        def f(x):
            calls.append(x)
            return x * 2

        assert f(1) == 2
        assert f(1) == 2
        assert calls == [1]

    def test_cache_clear(self):
        cache = cachetools.Cache(10)
        calls = []

        @cachetools.cached(cache)
        def f(x):
            calls.append(x)
            return x

        f(1)
        f.cache_clear()
        f(1)
        assert calls == [1, 1]

    def test_info(self):
        cache = cachetools.Cache(10)

        @cachetools.cached(cache, info=True)
        def f(x):
            return x

        f(1)
        f(1)
        info = f.cache_info()
        assert info.hits == 1
        assert info.misses == 1
        assert info.maxsize == 10

    def test_info_mapping(self):
        cache = {}

        @cachetools.cached(cache, info=True)
        def f(x):
            return x

        f(1)
        info = f.cache_info()
        assert info.maxsize is None

    def test_info_none_cache(self):
        import warnings
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")

            @cachetools.cached(None, info=True)
            def f(x):
                return x

        f(1)
        info = f.cache_info()
        assert info.hits == 0
        assert info.misses == 1

    def test_none_cache_deprecated(self):
        with pytest.warns(DeprecationWarning):
            @cachetools.cached(None)
            def f(x):
                return x

    def test_locked(self):
        import threading
        cache = cachetools.Cache(10)
        lock = threading.Lock()

        @cachetools.cached(cache, lock=lock)
        def f(x):
            return x

        assert f(1) == 1
        assert f(1) == 1

    def test_condition(self):
        import threading
        cache = cachetools.Cache(10)
        lock = threading.Lock()
        cond = threading.Condition(lock)

        @cachetools.cached(cache, lock=lock, condition=cond)
        def f(x):
            return x

        assert f(1) == 1
        assert f(1) == 1

    def test_cache_attributes(self):
        cache = cachetools.Cache(10)

        @cachetools.cached(cache)
        def f(x):
            return x

        assert f.cache is cache
        assert f.cache_info is None


class TestCachedMethod:
    def test_basic(self):
        class A:
            def __init__(self):
                self.cache = cachetools.Cache(10)
                self.calls = 0

            @cachetools.cachedmethod(lambda self: self.cache)
            def f(self, x):
                self.calls += 1
                return x * 2

        a = A()
        assert a.f(1) == 2
        assert a.f(1) == 2
        assert a.calls == 1

    def test_cache_clear(self):
        class A:
            def __init__(self):
                self.cache = cachetools.Cache(10)
                self.calls = 0

            @cachetools.cachedmethod(lambda self: self.cache)
            def f(self, x):
                self.calls += 1
                return x

        a = A()
        a.f(1)
        a.f.cache_clear()
        a.f(1)
        assert a.calls == 2

    def test_info(self):
        class A:
            def __init__(self):
                self.cache = cachetools.Cache(10)

            @cachetools.cachedmethod(lambda self: self.cache, info=True)
            def f(self, x):
                return x

        a = A()
        a.f(1)
        a.f(1)
        info = a.f.cache_info()
        assert info.hits == 1
        assert info.misses == 1

    def test_locked(self):
        import threading
        lock = threading.Lock()

        class A:
            def __init__(self):
                self.cache = cachetools.Cache(10)

            @cachetools.cachedmethod(lambda self: self.cache, lock=lambda self: lock)
            def f(self, x):
                return x

        a = A()
        assert a.f(1) == 1
        assert a.f(1) == 1

    def test_condition(self):
        import threading
        lock = threading.Lock()
        cond = threading.Condition(lock)

        class A:
            def __init__(self):
                self.cache = cachetools.Cache(10)

            @cachetools.cachedmethod(
                lambda self: self.cache,
                lock=lambda self: lock,
                condition=lambda self: cond,
            )
            def f(self, x):
                return x

        a = A()
        assert a.f(1) == 1
        assert a.f(1) == 1
